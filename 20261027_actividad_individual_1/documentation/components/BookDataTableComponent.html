import { CurrencyPipe } from '@angular/common';
import { ChangeDetectionStrategy, Component, computed, input, output, signal } from '@angular/core';
import { IonSearchbar, IonSpinner, IonText } from '@ionic/angular';
import { Libro } from '../../../core/models/database.models';

/** Columnas por las que se puede ordenar la tabla. */
export type BookSortKey = 'titulo' | 'autor' | 'categoria' | 'precio' | 'stock';
/** Sentido de la ordenación. */
export type SortDirection = 'asc' | 'desc';

/**
 * Tabla de libros reutilizable con buscador y ordenación.
 * Agnóstica del contenedor: recibe los datos por `libros` y notifica
 * las interacciones mediante `rowSelected` y `searchChange`.
 */
@Component({
  selector: 'app-book-data-table',
  templateUrl: './book-data-table.component.html',
  styleUrls: ['./book-data-table.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [CurrencyPipe, IonSearchbar, IonSpinner, IonText],
})
export class BookDataTableComponent {
  /** Libros a mostrar. */
  public readonly libros = input<Libro[]>([]);
  /** Muestra un indicador de carga en lugar de la tabla. */
  public readonly loading = input(false);
  /** Título accesible de la tabla (`<caption>`). */
  public readonly caption = input('Listado de libros');

  /** Se emite al pulsar una fila (clic o teclado). */
  public readonly rowSelected = output<Libro>();
  /** Se emite con el texto de búsqueda cada vez que cambia. */
  public readonly searchChange = output<string>();

  /** Texto de búsqueda actual. */
  protected readonly term = signal('');
  /** Columna de ordenación activa. */
  protected readonly sortKey = signal<BookSortKey>('titulo');
  /** Sentido de ordenación activo. */
  protected readonly sortDir = signal<SortDirection>('asc');

  /** Definición de columnas (clave, texto y alineación). */
  protected readonly columns: { key: BookSortKey; label: string; numeric: boolean }[] = [
    { key: 'titulo', label: 'Título', numeric: false },
    { key: 'autor', label: 'Autor', numeric: false },
    { key: 'categoria', label: 'Categoría', numeric: false },
    { key: 'precio', label: 'Precio', numeric: true },
    { key: 'stock', label: 'Stock', numeric: true },
  ];

  /** Filas filtradas por búsqueda y ordenadas. */
  protected readonly rows = computed(() => {
    const term = this.term().trim().toLowerCase();
    const key = this.sortKey();
    const factor = this.sortDir() === 'asc' ? 1 : -1;
    const filtered = term
      ? this.libros().filter((l) =>
          [l.titulo, l.autor, l.categoria, l.isbn ?? ''].some((v) => v.toLowerCase().includes(term)),
        )
      : this.libros();
    return [...filtered].sort((a, b) => {
      const x = a[key];
      const y = b[key];
      return (
        (typeof x === 'number' && typeof y === 'number'
          ? x - y
          : String(x).localeCompare(String(y), 'es')) * factor
      );
    });
  });

  /** Actualiza el término de búsqueda y emite `searchChange`. */
  protected onSearch(event: Event): void {
    const value = String((event as CustomEvent).detail.value ?? '');
    this.term.set(value);
    this.searchChange.emit(value);
  }

  /** Ordena por la columna dada; si ya estaba activa, invierte el sentido. */
  protected sortBy(key: BookSortKey): void {
    if (this.sortKey() === key) {
      this.sortDir.update((d) => (d === 'asc' ? 'desc' : 'asc'));
    } else {
      this.sortKey.set(key);
      this.sortDir.set('asc');
    }
  }

  /** Valor de `aria-sort` de una columna. */
  protected ariaSort(key: BookSortKey): 'ascending' | 'descending' | 'none' {
    if (this.sortKey() !== key) return 'none';
    return this.sortDir() === 'asc' ? 'ascending' : 'descending';
  }
}

<ion-searchbar
  class="book-table__search"
  placeholder="Buscar por título, autor, categoría o ISBN"
  aria-label="Buscar libros"
  [debounce]="250"
  (ionInput)="onSearch($event)"
></ion-searchbar>

@if (loading()) {
  <div class="book-table__state">
    <ion-spinner name="dots" aria-label="Cargando libros"></ion-spinner>
  </div>
} @else {
  <div class="book-table__scroll">
    <table class="book-table__table">
      <caption class="book-table__caption">
        {{ caption() }} ({{ rows().length }} {{ rows().length === 1 ? 'resultado' : 'resultados' }})
      </caption>
      <thead>
        <tr>
          @for (col of columns; track col.key) {
            <th
              scope="col"
              class="book-table__th"
              [class.book-table__th--numeric]="col.numeric"
              [attr.aria-sort]="ariaSort(col.key)"
            >
              <button type="button" class="book-table__sort" (click)="sortBy(col.key)">
                {{ col.label }}
                @if (sortKey() === col.key) {
                  <span aria-hidden="true">{{ sortDir() === 'asc' ? '▲' : '▼' }}</span>
                }
              </button>
            </th>
          }
        </tr>
      </thead>
      <tbody>
        @for (libro of rows(); track libro.id) {
          <tr
            class="book-table__row"
            tabindex="0"
            (click)="rowSelected.emit(libro)"
            (keydown.enter)="rowSelected.emit(libro)"
            (keydown.space)="rowSelected.emit(libro); $event.preventDefault()"
          >
            <th scope="row" class="book-table__title">{{ libro.titulo }}</th>
            <td>{{ libro.autor }}</td>
            <td>{{ libro.categoria }}</td>
            <td class="book-table__cell--numeric">{{ libro.precio | currency: 'EUR' }}</td>
            <td
              class="book-table__cell--numeric"
              [class.book-table__cell--low]="libro.stock <= 5"
            >
              {{ libro.stock }}
            </td>
          </tr>
        } @empty {
          <tr>
            <td colspan="5" class="book-table__empty">
              <ion-text color="medium">No hay libros que coincidan con la búsqueda.</ion-text>
            </td>
          </tr>
        }
      </tbody>
    </table>
  </div>
}
