import { CurrencyPipe } from '@angular/common';
import { ChangeDetectionStrategy, Component, computed, input, output } from '@angular/core';
import { Libro } from '../../../core/models/database.models';
import { BookCoverComponent } from '../book-cover/book-cover.component';

/**
 * Tarjeta de libro del catálogo: portada, valoración, precio con descuento y estado de stock.
 * Agnóstica del contenedor: recibe el libro por `libro` y emite `selected` al activarse.
 */
@Component({
  selector: 'app-book-card',
  templateUrl: './book-card.component.html',
  styleUrls: ['./book-card.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush,
  imports: [CurrencyPipe, BookCoverComponent],
})
export class BookCardComponent {
  /** Libro a mostrar. */
  public readonly libro = input.required<Libro>();

  /** Se emite al pulsar la tarjeta (clic o teclado). */
  public readonly selected = output<Libro>();

  /** Porcentaje de descuento (0 si no hay). */
  protected readonly descuento = computed(() => this.libro().descuento ?? 0);

  /** Precio final tras aplicar el descuento. */
  protected readonly precioFinal = computed(
    () => this.libro().precio * (1 - this.descuento() / 100),
  );

  /** Estado de stock con su texto y su variante visual. */
  protected readonly stock = computed(() => {
    const s = this.libro().stock;
    if (s === 0) return { text: 'Agotado', kind: 'out' };
    if (s <= 5) return { text: `Últimas ${s} uds.`, kind: 'low' };
    return { text: 'En stock', kind: 'ok' };
  });
}

<button type="button" class="book-card" (click)="selected.emit(libro())">
  <span class="book-card__cover">
    <app-book-cover [titulo]="libro().titulo" [autor]="libro().autor" />
    @if (descuento() > 0) {
      <span class="book-card__badge book-card__badge--discount">-{{ descuento() }}%</span>
    }
    @if (libro().destacado) {
      <span class="book-card__badge book-card__badge--featured">Destacado</span>
    }
  </span>

  <span class="book-card__body">
    <span class="book-card__category">{{ libro().categoria }}</span>
    <span class="book-card__title">{{ libro().titulo }}</span>
    <span class="book-card__author">{{ libro().autor }}</span>

    <span class="book-card__meta">
      @if (libro().valoracion; as v) {
        <span class="book-card__rating">
          <span class="sr-only">Valoración {{ v }} de 5</span>
          <span aria-hidden="true">★ {{ v }}</span>
        </span>
      }
      <span class="book-card__stock book-card__stock--{{ stock().kind }}">{{ stock().text }}</span>
    </span>

    <span class="book-card__price">
      <span class="book-card__price-now">{{ precioFinal() | currency: 'EUR' }}</span>
      @if (descuento() > 0) {
        <s class="book-card__price-old">{{ libro().precio | currency: 'EUR' }}</s>
      }
    </span>
  </span>
</button>
